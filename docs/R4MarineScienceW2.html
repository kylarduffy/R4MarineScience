Phase 1: Ingestion and Decontamination

Code 
# Load packages
library(tidyverse)
library(readxl)

# Import CSV datasets
metadata <- read_csv(
  here::here("data/workshop2/estuary_metadata.csv"))

sonde_data <- read_csv(
  here::here("data/workshop2/estuary_sonde_data.csv"))

species_dictionary <- read_csv(
  here::here("data/workshop2/species_dictionary.csv"))

catch_sheets <- excel_sheets(
  here::here("data/workshop2/estuary_catch_log.xlsx")) 

catch_sheets # view the sheet names

# Read all catch log tabs and combine them into one data frame
catch_data <- map_dfr( # combine all sheets into one data frame by rows
  catch_sheets,
  ~ read_excel(
    here::here("data/workshop2/estuary_catch_log.xlsx"),
    sheet = .x)) # .x is a placeholder for the sheet name

# String standardisation, site and species name
catch_data <- catch_data |>
  mutate(
    site = str_to_lower(site),
    site = str_replace_all(site, " ", "_"),
    species = str_to_lower(species),
    species = str_replace_all(species, " ", "_"))
unique(catch_data$site) # check that above worked
unique(catch_data$species) # check that above worked

# Clean metadata site names
metadata <- metadata |>
  mutate(
    site_name = str_to_lower(site_name),
    site_name = str_replace_all(site_name, " ", "_"))
unique(metadata$site_name) # check

# Check species dictionary
species_dictionary # names match already so no need to clean

# Temporal parsing
glimpse(sonde_data) # tells us what the data types are for each column, timestamp is character data
head(sonde_data$timestamp) # see format of timestamps

# Parse the timestamp
sonde_data <- sonde_data |>
  mutate(
    timestamp = dmy_hm(timestamp))
glimpse(sonde_data) # check that timestamp is now a date-time object

# Remove hardware errors
sum(sonde_data$turbidity == -999, na.rm = TRUE) # check how many -999 values are in the turbidity column
sonde_data <- sonde_data |>
  mutate(
    turbidity = na_if(turbidity, -999)) # replace -999 with NA
sum(sonde_data$turbidity == -999, na.rm = TRUE) # check if worked

Phase 2: Relational Architecture

Code 
# Summarise the water sensor data into a daily summary table 
sonde_data <- sonde_data |>
  mutate(
    date = floor_date(timestamp, unit = "day")) # create a day variable

# Create a daily summary table
sonde_daily <- sonde_data |> 
  group_by(site, date) |> # treats each site on each day as a group
  summarise(
    mean_temperature = mean(temperature, na.rm = TRUE), # daily mean temp
    mean_salinity = mean(salinity, na.rm = TRUE), # daily mean salinity
    .groups = "drop") # return finished table without keeping it grouped

# Taxonomic translation
catch_translated <- catch_data |>
  left_join(
    species_dictionary,
    by = c("species" = "common_name")) # add scientific names to the catch data
sum(is.na(catch_translated$scientific_name)) # check
catch_translated <- catch_translated |>
  select(-species) # remove the common name column

# The grand join
master_data <- catch_translated |>
  left_join(
    metadata,
    by = c("site" = "site_name")) # join catch data with spatial metadata
master_data <- master_data |>
  left_join(
    sonde_daily,
    by = c("site", "date")) # join daily water quality

Phase 3: Zero-Catch Framework

Code 
# Check structure of sampling days
master_data |>
  distinct(site, date) |>
  count(site) # tells us how many days of sampling were done at each site 

# Complete all species across every site and sampling day
master_data <- master_data |>
  complete(
    nesting(site, date),
    scientific_name
  ) # make sure all species appear for every site x day, if a species wasn't recorded complete() will create a missing row
glimpse(master_data) # check new rows were added

master_data |>
  summarise(
    missing_count = sum(is.na(count)),
    missing_zone = sum(is.na(zone)),
    missing_salinity = sum(is.na(mean_salinity))) # confirm NA counts

# Restore the spatial metadata for completed zero-catch rows
master_data <- master_data |>
  select(-lat, -lon, -zone) |>
  left_join(
    metadata,
    by = c("site" = "site_name"))
sum(is.na(master_data$zone)) # check

# Restore water quality data for completed zero-catch rows
master_data <- master_data |>
  select(-mean_temperature, -mean_salinity) |>
  left_join(
    sonde_daily,
    by = c("site", "date"))
sum(is.na(master_data$mean_salinity)) # check

# Replace missing catch values with zero
master_data <- master_data |>
  mutate(
    count = coalesce(count, 0)) # keep the existing count where there is one; if count is NA , use 0 instead
sum(is.na(master_data$count)) # check
